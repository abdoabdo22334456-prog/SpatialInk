cmake_minimum_required(VERSION 3.21)
project(UNI VERSION 0.8.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CLAP_WRAPPER_OUTPUT_NAME UNI CACHE STRING "Plugin filename")
set(CLAP_WRAPPER_DOWNLOAD_DEPENDENCIES OFF CACHE BOOL "Use pinned local SDKs")
set(CLAP_WRAPPER_COPY_AFTER_BUILD OFF CACHE BOOL "Do not install automatically")
set(CLAP_WRAPPER_BUILD_TESTS OFF CACHE BOOL "No unrelated wrapper examples")
if(NOT CLAP_WRAPPER_SOURCE_DIR OR NOT VST3_SDK_ROOT OR NOT CLAP_SDK_ROOT)
  message(FATAL_ERROR "Set CLAP_WRAPPER_SOURCE_DIR, VST3_SDK_ROOT and CLAP_SDK_ROOT")
endif()
add_subdirectory("${CLAP_WRAPPER_SOURCE_DIR}" clap-wrapper EXCLUDE_FROM_ALL)
add_library(UNI MODULE uni.c)
target_include_directories(UNI PRIVATE include)
target_compile_definitions(UNI PRIVATE SPATIALINK_VST3=1)
target_link_libraries(UNI PRIVATE user32 gdi32 m)
set(VST3_TARGET UNI)
target_add_vst3_wrapper(TARGET UNI OUTPUT_NAME UNI
  WINDOWS_FOLDER_VST3 TRUE BUNDLE_IDENTIFIER local.spatialink.orbit.vst3 BUNDLE_VERSION 0.8.0
  ASSET_OUTPUT_DIRECTORY "${SPATIALINK_OUTPUT_DIR}")
target_compile_definitions(UNI-clap-wrapper-vst3-lib PRIVATE SPATIALINK_LIMIT_SAMPLE_RATE=1)
set_target_properties(UNI PROPERTIES PREFIX "" OUTPUT_NAME UNI SUFFIX ".vst3")
if(MINGW)
  target_link_libraries(UNI PRIVATE ole32 uuid shell32 shlwapi winmm)
  target_link_options(UNI PRIVATE -static -static-libgcc -static-libstdc++)
endif()
